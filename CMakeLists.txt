cmake_minimum_required(VERSION 3.18)
project(rmvpe VERSION 0.1.0 LANGUAGES C CXX)
option(RMVPE_VULKAN "Build the ggml Vulkan backend" OFF)
option(RMVPE_METAL "Build the ggml Metal backend (unvalidated)" OFF)
option(RMVPE_CUDA "Build the ggml CUDA backend (unvalidated)" OFF)
option(RMVPE_BUILD_TESTS "Build native tests" ON)
option(RMVPE_CPU_AVX2 "Compile MSVC GRU with AVX2/FMA (disable for older CPU targets)" ON)
set(RMVPE_GGML_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml" CACHE PATH "Existing ggml v0.19.0 checkout")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>)
    if(NOT CMAKE_SYSTEM_PROCESSOR)
        set(CMAKE_SYSTEM_PROCESSOR AMD64)
    endif()
endif()
if(NOT TARGET ggml)
    set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GGML_VULKAN ${RMVPE_VULKAN} CACHE BOOL "" FORCE)
    set(GGML_METAL ${RMVPE_METAL} CACHE BOOL "" FORCE)
    set(GGML_CUDA ${RMVPE_CUDA} CACHE BOOL "" FORCE)
    if(EXISTS "${RMVPE_GGML_SOURCE_DIR}/CMakeLists.txt")
        add_subdirectory("${RMVPE_GGML_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ggml")
    else()
        include(FetchContent)
        FetchContent_Declare(ggml
            URL https://github.com/ggml-org/ggml/archive/refs/tags/v0.19.0.tar.gz
            URL_HASH SHA256=cfb6512adda2853e6500a7c5b23f326987cb4c723e9f8f93c6c5a7e7e4861648)
        FetchContent_MakeAvailable(ggml)
    endif()
endif()
if(RMVPE_VULKAN)
    include(cmake/Vulkan.cmake)
    rmvpe_vulkan()
endif()
add_library(rmvpe STATIC src/rmvpe.cpp src/audio.cpp src/gru.cpp)
add_library(rmvpe::rmvpe ALIAS rmvpe)
target_include_directories(rmvpe PUBLIC include PRIVATE third_party src)
find_package(Threads REQUIRED)
target_link_libraries(rmvpe PUBLIC ggml PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(rmvpe PRIVATE /W4)
    if(RMVPE_CPU_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
        set_source_files_properties(src/gru.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
        target_compile_definitions(rmvpe PRIVATE RMVPE_GRU_AVX2)
    endif()
else()
    target_compile_options(rmvpe PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rmvpe-cli tools/rmvpe-cli.cpp)
target_link_libraries(rmvpe-cli PRIVATE rmvpe::rmvpe)
if(RMVPE_BUILD_TESTS)
    enable_testing()
    add_executable(rmvpe-tests tests/test_rmvpe.cpp)
    target_link_libraries(rmvpe-tests PRIVATE rmvpe::rmvpe)
    add_test(NAME rmvpe-unit COMMAND rmvpe-tests)
    add_executable(rmvpe-gru-tests tests/test_gru.cpp)
    target_include_directories(rmvpe-gru-tests PRIVATE src)
    target_link_libraries(rmvpe-gru-tests PRIVATE rmvpe::rmvpe)
    add_test(NAME rmvpe-gru-cpu COMMAND rmvpe-gru-tests CPU)
    add_executable(rmvpe-session-tests tests/test_session.cpp)
    target_link_libraries(rmvpe-session-tests PRIVATE rmvpe::rmvpe)
    set(RMVPE_TEST_MODEL "" CACHE FILEPATH "Optional converted GGUF for session tests")
    if(RMVPE_TEST_MODEL)
        add_test(NAME rmvpe-session-cpu COMMAND rmvpe-session-tests "${RMVPE_TEST_MODEL}" cpu)
    endif()
    set(RMVPE_TEST_BACKEND "" CACHE STRING "Optional device for GRU validation, e.g. Vulkan0")
    if(RMVPE_TEST_BACKEND)
        add_test(NAME rmvpe-gru-accelerator COMMAND rmvpe-gru-tests "${RMVPE_TEST_BACKEND}")
        if(RMVPE_TEST_MODEL)
            add_test(NAME rmvpe-session-accelerator COMMAND rmvpe-session-tests "${RMVPE_TEST_MODEL}" "${RMVPE_TEST_BACKEND}")
        endif()
    endif()
endif()
include(GNUInstallDirs)
install(TARGETS rmvpe rmvpe-cli ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/rmvpe DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE NOTICE.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/rmvpe)
install(DIRECTORY licenses DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/rmvpe)
